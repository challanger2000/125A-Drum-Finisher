cmake_minimum_required(VERSION 3.25)
project(DrumFinisher VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(drum_core_tests tests/DrumCoreTests.cpp)
target_include_directories(drum_core_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_sources(drum_core_tests PRIVATE src/dsp/AdaptiveResonanceSuppressor.cpp)
if(MSVC)
    target_compile_options(drum_core_tests PRIVATE /W4)
else()
    target_compile_options(drum_core_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME drum_core_contract COMMAND drum_core_tests)
add_executable(drum_dsp_qa tests/DrumDSPQA.cpp)
target_include_directories(drum_dsp_qa PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)

add_test(NAME drum_characterization COMMAND drum_characterization)

option(DRUM_BUILD_VST3 "Build VST3 using an existing recursively cloned Steinberg SDK" OFF)
if(DRUM_BUILD_VST3)
    if(NOT DEFINED VST3_SDK_ROOT OR NOT EXISTS "${VST3_SDK_ROOT}/CMakeLists.txt")
        message(FATAL_ERROR "VST3_SDK_ROOT must point to an existing Steinberg SDK 3.8.1 checkout")
    endif()
    set(SMTG_ENABLE_VSTGUI_SUPPORT ON CACHE BOOL "" FORCE)
    add_subdirectory("${VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/vst3sdk")
    smtg_add_vst3plugin(125A_Drum_Finisher src/vst/DrumPlugin.cpp src/DrumCore.h src/dsp/AdaptiveResonanceSuppressor.cpp src/gui/SteelKnob.cpp src/gui/SteelKnob.h)
    target_include_directories(125A_Drum_Finisher PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
    target_include_directories(125A_Drum_Finisher PRIVATE ${VST3_SDK_ROOT}/vstgui4)
    smtg_target_add_plugin_resources(125A_Drum_Finisher RESOURCES
        resource/DrumFinisher.uidesc
        resource/125A_FinisherRing_S_64px_100pct.png
        resource/125A_FinisherRing_S_96px_150pct.png
        resource/125A_FinisherRing_M_96px_100pct.png
        resource/125A_FinisherRing_M_144px_150pct.png
        resource/125A_FinisherRing_H_128px_100pct.png
        resource/125A_FinisherRing_H_192px_150pct.png)
    target_compile_features(125A_Drum_Finisher PRIVATE cxx_std_17)
    target_link_libraries(125A_Drum_Finisher PRIVATE sdk vstgui_support)
    if(WIN32)
        target_sources(125A_Drum_Finisher PRIVATE
            "${VST3_SDK_ROOT}/public.sdk/source/main/dllmain.cpp")
    endif()
endif()
